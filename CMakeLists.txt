cmake_minimum_required(VERSION 3.24)
project(PostmortemTFC VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Build with the x64 Microsoft C++ toolchain and Windows SDK.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(SPDLOG_INSTALL ON CACHE BOOL "" FORCE)
# OVERRIDE_FIND_PACKAGE makes CommonLib's find_package resolve this vendored target.
include(FetchContent)
FetchContent_Declare(spdlog SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/spdlog" OVERRIDE_FIND_PACKAGE)
FetchContent_MakeAvailable(spdlog)
set(RAPIDCSV_INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/rapidcsv/src" CACHE PATH "" FORCE)
add_subdirectory(ThirdParty/CommonLibSSE-NG)
add_library(PostmortemTFC SHARED src/main.cpp)
target_link_libraries(PostmortemTFC PRIVATE CommonLibSSE::CommonLibSSE user32)
target_compile_definitions(PostmortemTFC PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(PostmortemTFC PRIVATE /W4 /permissive- /utf-8)
set_target_properties(PostmortemTFC PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
install(TARGETS PostmortemTFC RUNTIME DESTINATION SKSE/Plugins)
install(FILES config/PostmortemTFC.ini DESTINATION SKSE/Plugins)
enable_testing()
add_executable(MotionTests tests/MotionTests.cpp)
target_include_directories(MotionTests PRIVATE src)
add_test(NAME CameraMotion COMMAND MotionTests)

add_executable(OrbitTests tests/OrbitTests.cpp)
target_include_directories(OrbitTests PRIVATE src)
add_test(NAME OrbitRotation COMMAND OrbitTests)
