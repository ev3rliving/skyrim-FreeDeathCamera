cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0141 NEW)
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
project(PostmortemTFC VERSION 1.2.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Build with the x64 Microsoft C++ toolchain and Windows SDK.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
include(cmake/ModernDependencies.cmake)
add_library(PostmortemTFC SHARED src/main.cpp)
target_link_libraries(PostmortemTFC PRIVATE CommonLibSSE::CommonLibSSE user32)
target_compile_definitions(PostmortemTFC PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(PostmortemTFC PRIVATE /W4 /permissive- /utf-8)
set_target_properties(PostmortemTFC PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
install(TARGETS PostmortemTFC RUNTIME DESTINATION SKSE/Plugins COMPONENT Camera)
install(FILES config/PostmortemTFC.ini DESTINATION SKSE/Plugins COMPONENT Camera)
enable_testing()
add_executable(MotionTests tests/MotionTests.cpp)
target_include_directories(MotionTests PRIVATE src)
add_test(NAME CameraMotion COMMAND MotionTests)

add_executable(OrbitTests tests/OrbitTests.cpp)
target_include_directories(OrbitTests PRIVATE src)
add_test(NAME OrbitRotation COMMAND OrbitTests)
add_executable(CompatibilityTests tests/CompatibilityTests.cpp)
target_include_directories(CompatibilityTests PRIVATE src)
add_test(NAME Compatibility COMMAND CompatibilityTests)
add_executable(RuntimeLayoutTests tests/RuntimeLayoutTests.cpp)
target_link_libraries(RuntimeLayoutTests PRIVATE CommonLibSSE::CommonLibSSE)
target_compile_definitions(RuntimeLayoutTests PRIVATE ENABLE_COMMONLIBSSE_TESTING=1)
add_test(NAME RuntimeLayouts COMMAND RuntimeLayoutTests)
add_test(NAME StartupInstallation COMMAND "${CMAKE_COMMAND}"
    "-DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}/src"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/StartupInstallation.cmake")


install(FILES README.md VERIFICATION.md CHANGELOG.md COPYING.txt EXCEPTIONS.md DESTINATION Docs/FreeDeathCamera COMPONENT Camera)
install(DIRECTORY licenses/ DESTINATION Docs/FreeDeathCamera/licenses COMPONENT Camera)
